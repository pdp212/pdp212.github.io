enzyme_action.py — Manim Scene 
from manim import *

class CinematicTitle(Scene):
    def construct(self):
        title = Text(
            "PHAN ĐỨC PHÁT",
            font="Playfair Display",
            color="#C5A880"
        )
        line = Line(LEFT*3, RIGHT*3)
        line.set_color("#8E8E93")

        self.play(Write(title), run_time=2)
        self.play(Create(line))
        self.wait(1)
